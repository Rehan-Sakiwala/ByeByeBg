## 7️⃣ SERVICE IMPLEMENTATION

## File: service/impl/TravelPlanServiceImpl.java

@Service
@RequiredArgsConstructor
public class TravelPlanServiceImpl implements TravelPlanService {

```
private final TravelPlanRepository travelPlanRepository;
private final EmployeeRepository employeeRepository;
private final TravelPlanValidator validator;
private final TravelPlanMapper mapper;

@Override
public TravelPlanResponseDto create(TravelPlanCreateRequestDto dto, Long employeeId) {

    Employee creator = employeeRepository.findById(employeeId)
            .orElseThrow(() -> new ResourceNotFoundException("Employee not found"));

    if(!creator.getRole().getName().equals("HR"))
        throw new BusinessException("Only HR can create travel plans");

    validator.validateDates(dto.getDepartureDate(), dto.getReturnDate());

    TravelPlan plan = new TravelPlan();
    plan.setTitle(dto.getTitle());
    plan.setDescription(dto.getDescription());
    plan.setDestination(dto.getDestination());
    plan.setDepartureDate(dto.getDepartureDate());
    plan.setReturnDate(dto.getReturnDate());
    plan.setCreatedBy(creator);

    travelPlanRepository.save(plan);

    return mapper.toDto(plan);
}

@Override
public List<TravelPlanResponseDto> getAllActive() {
    return travelPlanRepository.findByCancelledFalseOrderByDepartureDateAsc()
            .stream()
            .map(mapper::toDto)
            .toList();
}

@Override
public TravelPlanResponseDto cancel(Long planId, Long employeeId) {

    Employee emp = employeeRepository.findById(employeeId)
            .orElseThrow(() -> new ResourceNotFoundException("Employee not found"));

    if(!emp.getRole().getName().equals("HR"))
        throw new BusinessException("Only HR can cancel travel plans");

    TravelPlan plan = travelPlanRepository.findById(planId)
            .orElseThrow(() -> new ResourceNotFoundException("Travel plan not found"));

    plan.setCancelled(true);
    travelPlanRepository.save(plan);

    return mapper.toDto(plan);
}
```

}
